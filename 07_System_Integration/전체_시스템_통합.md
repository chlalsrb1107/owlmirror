```python
import threading
import queue
import time
import numpy as np

class VehicleAlertSystem:
    def __init__(self):
        # 컴포넌트 초기화
        self.preprocessor = AudioPreprocessor()
        self.classifier = TRTInference("sound_classifier.engine")
        self.doa_tracker = DoAKalmanTracker()
        self.camera = FisheyeTracker()
        self.video_pipeline = VideoPipeline(self._on_frame)

        # 공유 상태
        self.current_label = "background"
        self.current_doa = 0.0
        self.current_conf = 0.0
        self.current_frame = None
        self.lock = threading.Lock()

        # 스레드 큐
        self.audio_queue = queue.Queue(maxsize=4)
        self.display_queue = queue.Queue(maxsize=2)

    # ── 오디오 스레드 ──────────────────────────────────
    def audio_thread(self):
        stream = StreamProcessor(
            preprocessor=self.preprocessor,
            callback=self._on_audio,
            sr=16000, channels=6, block_size=512
        )
        stream.run()

    def _on_audio(self, tensor, raw_multichannel):
        # 1. 음원 분류
        probs = self.classifier.infer(tensor)
        label_idx = np.argmax(probs)
        labels = ['car_horn', 'emergency_siren',
                  'motorcycle_exhaust', 'background']
        label = labels[label_idx]
        conf = probs[label_idx]

        thresholds = {'car_horn': 0.70, 'emergency_siren': 0.65,
                      'motorcycle_exhaust': 0.75}
        if label != 'background' and conf < thresholds[label]:
            label = 'background'

        # 2. DoA 추정 (배경이 아닐 때만)
        if label != 'background':
            tdoa = compute_all_tdoas(raw_multichannel)
            raw_doa = tdoa_to_doa(tdoa, mic_positions)
            smooth_doa = self.doa_tracker.update(raw_doa)
        else:
            smooth_doa = self.current_doa

        with self.lock:
            self.current_label = label
            self.current_doa = smooth_doa
            self.current_conf = conf

    # ── 영상 스레드 ──────────────────────────────────
    def _on_frame(self, frame):
        with self.lock:
            label = self.current_label
            doa = self.current_doa
            conf = self.current_conf

        roi = self.camera.get_roi_frame(frame, doa)
        output = render_hud(roi, label, doa, conf)

        if not self.display_queue.full():
            self.display_queue.put(output)

    # ── 디스플레이 스레드 ──────────────────────────────
    def display_thread(self):
        cv2.namedWindow('Driver Display', cv2.WND_PROP_FULLSCREEN)
        cv2.setWindowProperty('Driver Display',
                               cv2.WND_PROP_FULLSCREEN,
                               cv2.WINDOW_FULLSCREEN)
        while True:
            try:
                frame = self.display_queue.get(timeout=0.1)
                cv2.imshow('Driver Display', frame)
                if cv2.waitKey(1) & 0xFF == ord('q'):
                    break
            except queue.Empty:
                pass

    # ── 시스템 시작 ────────────────────────────────────
    def run(self):
        threads = [
            threading.Thread(target=self.audio_thread, daemon=True),
            threading.Thread(target=self.display_thread, daemon=True),
        ]
        for t in threads:
            t.start()
        self.video_pipeline.start()
        for t in threads:
            t.join()

if __name__ == "__main__":
    system = VehicleAlertSystem()
    system.run()
```

## 성능 최적화 포인트

| 항목 | 최적화 방법 |
|---|---|
| 오디오 추론 | TensorRT FP16, 배치 크기 1 고정 |
| GCC-PHAT | NumPy vectorize, CUDA FFT (cupy) 적용 가능 |
| 영상 처리 | NVENC 하드웨어 인코딩, nvvidconv 사용 |
| 메모리 | 제로카피 NVMM 메모리 활용 |
| CPU 스케줄링 | 오디오/영상 스레드 core affinity 분리 |

## 시스템 부팅 자동 시작

```bash
# /etc/systemd/system/vehicle-alert.service
[Unit]
Description=Vehicle Alert System
After=sound.target graphical.target

[Service]
ExecStart=/usr/bin/python3 /opt/vehicle-alert/main.py
Restart=always
User=user
Environment=DISPLAY=:0

[Install]
WantedBy=graphical.target
```

```bash
sudo systemctl enable vehicle-alert
sudo systemctl start vehicle-alert
```
