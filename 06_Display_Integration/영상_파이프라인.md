## Python 통합 파이프라인

```python
import gi
gi.require_version('Gst', '1.0')
from gi.repository import Gst, GLib
import numpy as np
import cv2
import threading

Gst.init(None)

class VideoPipeline:
    def __init__(self, on_frame_callback):
        self.callback = on_frame_callback
        pipeline_str = (
            "nvarguscamerasrc sensor-id=0 ! "
            "video/x-raw(memory:NVMM),width=1920,height=1080,framerate=30/1 ! "
            "nvvidconv ! video/x-raw,format=BGRx ! "
            "videoconvert ! video/x-raw,format=BGR ! "
            "appsink name=sink max-buffers=2 drop=true emit-signals=true"
        )
        self.pipeline = Gst.parse_launch(pipeline_str)
        sink = self.pipeline.get_by_name('sink')
        sink.connect('new-sample', self._on_new_sample)

    def _on_new_sample(self, sink):
        sample = sink.emit('pull-sample')
        buf = sample.get_buffer()
        caps = sample.get_caps()
        h = caps.get_structure(0).get_value('height')
        w = caps.get_structure(0).get_value('width')
        success, mapinfo = buf.map(Gst.MapFlags.READ)
        frame = np.frombuffer(mapinfo.data, dtype=np.uint8).reshape(h, w, 3)
        buf.unmap(mapinfo)
        self.callback(frame.copy())
        return Gst.FlowReturn.OK

    def start(self):
        self.pipeline.set_state(Gst.State.PLAYING)

    def stop(self):
        self.pipeline.set_state(Gst.State.NULL)
```

## HUD 오버레이 렌더링

```python
import cv2
import numpy as np

LABEL_COLORS = {
    'car_horn':           (0, 255, 255),   # 노란색
    'emergency_siren':    (0, 0, 255),     # 빨간색
    'motorcycle_exhaust': (0, 165, 255),   # 주황색
    'background':         (128, 128, 128), # 회색
}

def render_hud(frame: np.ndarray, label: str,
               doa: float, confidence: float) -> np.ndarray:
    if label == 'background':
        return frame

    color = LABEL_COLORS.get(label, (255, 255, 255))
    h, w = frame.shape[:2]

    # 경보 배너
    cv2.rectangle(frame, (0, 0), (w, 70), (0, 0, 0), -1)
    cv2.putText(frame, f"[ALERT] {label.replace('_', ' ').upper()}",
                (20, 35), cv2.FONT_HERSHEY_DUPLEX, 1.0, color, 2)
    cv2.putText(frame, f"Direction: {doa:.0f}  |  Conf: {confidence:.0%}",
                (20, 62), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (200, 200, 200), 1)

    # 방향 나침반 (우하단)
    cx, cy = w - 80, h - 80
    cv2.circle(frame, (cx, cy), 50, (50, 50, 50), -1)
    cv2.circle(frame, (cx, cy), 50, color, 2)
    # 방향 화살표
    rad = np.radians(doa - 90)  # 전방 = 위
    ax = int(cx + 40 * np.cos(rad))
    ay = int(cy + 40 * np.sin(rad))
    cv2.arrowedLine(frame, (cx, cy), (ax, ay), color, 3, tipLength=0.3)
    cv2.putText(frame, "N", (cx - 5, cy - 55),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (200, 200, 200), 1)

    return frame
```

## HDMI 출력

```python
# OpenCV 전체 화면 출력
cv2.namedWindow('Driver Display', cv2.WND_PROP_FULLSCREEN)
cv2.setWindowProperty('Driver Display',
                       cv2.WND_PROP_FULLSCREEN,
                       cv2.WINDOW_FULLSCREEN)

while True:
    frame = get_current_frame()
    label, doa, conf = get_detection_result()
    frame = render_hud(frame, label, doa, conf)
    cv2.imshow('Driver Display', frame)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break
```
