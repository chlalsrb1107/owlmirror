# 오디오 전처리 파이프라인

## 개요

```
6ch PCM 입력
    → 채널 믹스다운 (1ch)
    → 하이패스 필터 (엔진음 제거)
    → VAD (음성 활동 감지 → 소음 제거)
    → Mel-Spectrogram 변환
    → 정규화
    → 모델 입력 텐서
```

---

## 구현 코드

```python
import numpy as np
import librosa
from scipy.signal import butter, lfilter

class AudioPreprocessor:
    def __init__(self, sr=16000, n_mels=128, hop_length=512, n_fft=1024):
        self.sr = sr
        self.n_mels = n_mels
        self.hop_length = hop_length
        self.n_fft = n_fft
        # 엔진음 제거용 하이패스 필터 (300Hz 이상 통과)
        self.b, self.a = butter(4, 300 / (sr / 2), btype='high')

    def mix_down(self, multichannel: np.ndarray) -> np.ndarray:
        # (6, N) → (N,) 평균 믹스다운
        return multichannel.mean(axis=0)

    def highpass_filter(self, y: np.ndarray) -> np.ndarray:
        return lfilter(self.b, self.a, y)

    def to_mel_spectrogram(self, y: np.ndarray) -> np.ndarray:
        mel = librosa.feature.melspectrogram(
            y=y, sr=self.sr,
            n_mels=self.n_mels,
            hop_length=self.hop_length,
            n_fft=self.n_fft,
            fmin=300,    # 엔진음 아래 컷
            fmax=8000    # 경적/사이렌 주파수 상한
        )
        log_mel = librosa.power_to_db(mel, ref=np.max)
        return log_mel  # shape: (n_mels, time_frames)

    def normalize(self, spec: np.ndarray) -> np.ndarray:
        # [-80dB, 0dB] → [0, 1] 정규화
        return (spec + 80) / 80

    def process(self, multichannel: np.ndarray) -> np.ndarray:
        y = self.mix_down(multichannel)
        y = self.highpass_filter(y)
        spec = self.to_mel_spectrogram(y)
        spec = self.normalize(spec)
        # 고정 길이 패딩/트리밍 (128 frames = ~1.6초 @ 16kHz)
        target_frames = 128
        if spec.shape[1] < target_frames:
            spec = np.pad(spec, ((0,0),(0, target_frames - spec.shape[1])))
        else:
            spec = spec[:, :target_frames]
        return spec[np.newaxis, np.newaxis, :, :]  # (1, 1, 128, 128)
```

## 실시간 스트리밍 처리

```python
import sounddevice as sd
import queue
import threading

class StreamProcessor:
    def __init__(self, preprocessor, callback, device_id=None,
                 sr=16000, channels=6, block_size=512):
        self.preprocessor = preprocessor
        self.callback = callback
        self.queue = queue.Queue()
        self.buffer = np.zeros((channels, sr * 2))  # 2초 링 버퍼

        self.stream = sd.InputStream(
            device=device_id,
            samplerate=sr,
            channels=channels,
            blocksize=block_size,
            dtype='int16',
            callback=self._audio_callback
        )

    def _audio_callback(self, indata, frames, time, status):
        # indata: (block_size, channels) → 전치 후 큐에 추가
        self.queue.put(indata.T.astype(np.float32) / 32768.0)

    def run(self):
        with self.stream:
            while True:
                chunk = self.queue.get()
                # 링 버퍼 업데이트
                self.buffer = np.roll(self.buffer, -chunk.shape[1], axis=1)
                self.buffer[:, -chunk.shape[1]:] = chunk
                # 전처리 및 추론 콜백
                tensor = self.preprocessor.process(self.buffer)
                self.callback(tensor, self.buffer)
```
