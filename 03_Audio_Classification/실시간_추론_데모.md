# 실시간 추론 데모 — ReSpeaker → 스펙트로그램 → 모델 → 경보

## 전체 흐름 한눈에 보기

```
ReSpeaker 마이크
    │
    │ 매 32ms마다 512 샘플 블록 전달
    ▼
[링 버퍼] ← 항상 최근 2초치 오디오를 보관
    │
    │ 0.5초마다 버퍼 스냅샷 꺼냄
    ▼
[전처리]
    ① 4ch → 1ch 믹스다운 (평균)
    ② 하이패스 필터 (300Hz 이하 엔진음 제거)
    ③ Mel-Spectrogram 변환 → (128 mel bands × 128 time frames) 행렬
    ④ dB 정규화 → [0, 1] 픽셀값처럼 정규화
    │
    │ (1, 1, 128, 128) 텐서
    ▼
[TensorRT 모델 추론]
    │ 4개 클래스 확률 출력
    │ [car_horn, emergency_siren, motorcycle_exhaust, background]
    ▼
[경보 판단]
    confidence > 임계값 AND 클래스 ≠ background
    → True  : 화면 + 콘솔 경보 출력
    → False : 아무 반응 없음
```

---

## 왜 스펙트로그램인가

소리는 시간에 따라 변하는 1D 파형이다. 이걸 CNN에 직접 넣기는 어렵다. 스펙트로그램은 **"시간 × 주파수 × 에너지"** 를 2D 이미지로 표현한 것이라 CNN이 이미지 분류하듯 패턴을 학습할 수 있다.

```
경적 소리의 Mel-Spectrogram 모습:
높은 에너지  ████████░░░░░░░░  ← 경적 주파수 대역 (500~3000Hz)
             시간 →
낮은 에너지  ░░░░░░░░░░░░░░░░  ← 낮은 주파수 대역 (엔진음, 필터링됨)

사이렌 소리:
높은 에너지  ░██░░██░░██░░██░  ← 주파수가 위아래로 스윕하는 패턴
```

## Step 1: 스펙트로그램 변환 원리

```python
import numpy as np
import librosa
import matplotlib.pyplot as plt
from scipy.signal import butter, lfilter

def show_spectrogram_pipeline(audio_file: str):
    """
    오디오 파일 하나를 불러서 각 단계를 시각적으로 보여주는 함수.
    처음 이해할 때 실행해보면 좋다.
    """
    # 1. 오디오 로드
    y, sr = librosa.load(audio_file, sr=16000, mono=True)
    print(f"원본 파형: {len(y)} 샘플, {len(y)/sr:.2f}초")

    # 2. 하이패스 필터 (엔진음 제거)
    b, a = butter(4, 300 / (sr / 2), btype='high')
    y_filtered = lfilter(b, a, y)

    # 3. Mel-Spectrogram 변환
    mel = librosa.feature.melspectrogram(
        y=y_filtered,
        sr=sr,
        n_mels=128,       # 주파수 축 해상도 (128개 mel 대역)
        hop_length=256,   # 몇 샘플마다 FFT 슬라이딩 (작을수록 시간 해상도 높음)
        n_fft=1024,       # FFT 창 크기
        fmin=300,         # 최소 주파수 (엔진음 아래 제거)
        fmax=8000         # 최대 주파수 (경적/사이렌 상한)
    )

    # 4. dB 변환 + 정규화
    log_mel = librosa.power_to_db(mel, ref=np.max)  # dB 스케일 (0 ~ -80dB)
    normalized = (log_mel + 80) / 80                 # 0~1 범위로 정규화

    print(f"스펙트로그램 shape: {normalized.shape}")
    # → (128, time_frames) — 128개 주파수 대역 × 시간 프레임 수

    # 시각화
    fig, axes = plt.subplots(2, 2, figsize=(12, 8))
    axes[0, 0].plot(y[:sr])
    axes[0, 0].set_title("원본 파형 (1초)")

    axes[0, 1].plot(y_filtered[:sr])
    axes[0, 1].set_title("하이패스 필터 후 (엔진음 제거)")

    librosa.display.specshow(log_mel, sr=sr, hop_length=256,
                              x_axis='time', y_axis='mel', ax=axes[1, 0])
    axes[1, 0].set_title("Mel-Spectrogram (dB)")

    axes[1, 1].imshow(normalized, aspect='auto', origin='lower',
                       cmap='viridis', vmin=0, vmax=1)
    axes[1, 1].set_title("정규화 후 (모델 입력)")
    axes[1, 1].set_xlabel("Time frames")
    axes[1, 1].set_ylabel("Mel bands")

    plt.tight_layout()
    plt.savefig("spectrogram_pipeline.png")
    plt.show()

# 사용 예시:
# show_spectrogram_pipeline("car_horn_sample.wav")
```

## Step 2: 실시간 스트리밍 + 경보 (완전한 실행 코드)

이 파일을 Jetson에서 직접 실행하면 ReSpeaker로부터 음성을 받아 추론하고, 경적/사이렌/배기음이 감지되면 경보를 출력한다.

```python
"""
realtime_inference.py
실행 방법:
    python3 realtime_inference.py --model sound_classifier.engine
    python3 realtime_inference.py --model sound_classifier.pt  # PyTorch 버전
"""

import argparse
import threading
import queue
import time
import sys

import numpy as np
import sounddevice as sd
from scipy.signal import butter, lfilter
import librosa

# ── 설정 ───────────────────────────────────────────────────────────
SR          = 16000    # 샘플링 레이트
N_MELS      = 128      # Mel 대역 수
HOP_LENGTH  = 256      # STFT 슬라이딩 간격
N_FFT       = 1024     # FFT 창 크기
TARGET_FRAMES = 128    # 스펙트로그램 시간 프레임 수 (= 약 2초)
BUFFER_SEC  = 2.0      # 링 버퍼 길이 (초)
INFER_EVERY = 0.5      # 추론 주기 (초)
CHANNELS    = 4        # ReSpeaker v3.0 채널 수

LABELS = ['car_horn', 'emergency_siren', 'motorcycle_exhaust', 'background']
THRESHOLDS = {
    'car_horn':           0.70,
    'emergency_siren':    0.65,
    'motorcycle_exhaust': 0.75,
}
ALERT_MESSAGES = {
    'car_horn':           "⚠️  경적 감지!",
    'emergency_siren':    "🚨 긴급차량 사이렌 감지!",
    'motorcycle_exhaust': "🏍️  오토바이 접근!",
}

# ── 전처리 ─────────────────────────────────────────────────────────
_b, _a = butter(4, 300 / (SR / 2), btype='high')

def preprocess(multichannel: np.ndarray) -> np.ndarray:
    """
    (4, N) float32 배열 → (1, 1, 128, 128) 텐서
    """
    # ① 4채널 → 1채널 믹스다운
    mono = multichannel.mean(axis=0)

    # ② 하이패스 필터 (엔진음 300Hz 이하 제거)
    mono = lfilter(_b, _a, mono)

    # ③ Mel-Spectrogram 변환
    mel = librosa.feature.melspectrogram(
        y=mono, sr=SR,
        n_mels=N_MELS, hop_length=HOP_LENGTH,
        n_fft=N_FFT, fmin=300, fmax=8000
    )

    # ④ dB 변환 + 정규화 → [0, 1]
    log_mel = librosa.power_to_db(mel, ref=np.max)
    normalized = (log_mel + 80) / 80
    normalized = np.clip(normalized, 0, 1)

    # ⑤ 고정 길이 맞추기 (패딩 또는 트리밍)
    if normalized.shape[1] < TARGET_FRAMES:
        pad = TARGET_FRAMES - normalized.shape[1]
        normalized = np.pad(normalized, ((0, 0), (0, pad)))
    else:
        normalized = normalized[:, :TARGET_FRAMES]

    # ⑥ 모델 입력 shape: (batch=1, channel=1, H=128, W=128)
    return normalized[np.newaxis, np.newaxis, :, :].astype(np.float32)


# ── 모델 로더 ──────────────────────────────────────────────────────
def load_model(model_path: str):
    """
    .engine (TensorRT) 또는 .pt (PyTorch) 자동 감지 후 로드.
    """
    if model_path.endswith('.engine'):
        return _load_trt(model_path)
    else:
        return _load_torch(model_path)

def _load_trt(engine_path: str):
    import tensorrt as trt
    import pycuda.driver as cuda
    import pycuda.autoinit

    logger = trt.Logger(trt.Logger.WARNING)
    with open(engine_path, 'rb') as f:
        engine = trt.Runtime(logger).deserialize_cuda_engine(f.read())
    context = engine.create_execution_context()

    # 입/출력 버퍼 미리 할당
    in_shape  = (1, 1, 128, 128)
    out_shape = (1, len(LABELS))
    h_in   = cuda.pagelocked_empty(in_shape,  dtype=np.float32)
    h_out  = cuda.pagelocked_empty(out_shape, dtype=np.float32)
    d_in   = cuda.mem_alloc(h_in.nbytes)
    d_out  = cuda.mem_alloc(h_out.nbytes)
    stream = cuda.Stream()

    def infer_trt(tensor: np.ndarray) -> np.ndarray:
        np.copyto(h_in, tensor)
        cuda.memcpy_htod_async(d_in, h_in, stream)
        context.execute_async_v2(
            bindings=[int(d_in), int(d_out)], stream_handle=stream.handle
        )
        cuda.memcpy_dtoh_async(h_out, d_out, stream)
        stream.synchronize()
        # 소프트맥스 적용
        e = np.exp(h_out[0] - h_out[0].max())
        return e / e.sum()

    return infer_trt

def _load_torch(pt_path: str):
    import torch
    model = torch.load(pt_path, map_location='cuda')
    model.eval()

    def infer_torch(tensor: np.ndarray) -> np.ndarray:
        with torch.no_grad():
            t = torch.from_numpy(tensor).cuda()
            logits = model(t)
            probs = torch.softmax(logits, dim=1).cpu().numpy()[0]
        return probs

    return infer_torch


# ── 링 버퍼 ────────────────────────────────────────────────────────
class RingBuffer:
    """
    멀티채널 오디오를 BUFFER_SEC 길이로 계속 덮어쓰는 원형 버퍼.
    항상 가장 최근 2초치 데이터만 유지한다.
    """
    def __init__(self, channels: int, seconds: float, sr: int):
        self.size = int(seconds * sr)
        self.buf  = np.zeros((channels, self.size), dtype=np.float32)
        self.lock = threading.Lock()

    def write(self, chunk: np.ndarray):
        # chunk shape: (channels, n_samples)
        n = chunk.shape[1]
        with self.lock:
            self.buf = np.roll(self.buf, -n, axis=1)
            self.buf[:, -n:] = chunk

    def read(self) -> np.ndarray:
        with self.lock:
            return self.buf.copy()


# ── 메인 시스템 ────────────────────────────────────────────────────
class SoundAlertSystem:
    def __init__(self, model_path: str, device_id=None):
        print("모델 로딩 중...")
        self.infer = load_model(model_path)
        self.ring  = RingBuffer(CHANNELS, BUFFER_SEC, SR)
        self.device_id = device_id
        self._running = False
        self._last_alert_time = 0
        self._alert_cooldown  = 2.0  # 같은 경보 연속 출력 방지 (초)
        print("준비 완료. 음성 감지 시작...\n")

    # ── 오디오 콜백 (sounddevice가 자동으로 호출) ──────────────────
    def _audio_callback(self, indata, frames, time_info, status):
        # indata shape: (frames, channels) → 전치해서 (channels, frames)
        chunk = indata.T.astype(np.float32) / 32768.0
        self.ring.write(chunk)

    # ── 추론 스레드 ────────────────────────────────────────────────
    def _inference_loop(self):
        while self._running:
            start = time.time()

            # 1. 링 버퍼에서 최근 2초 오디오 가져오기
            audio = self.ring.read()         # (4, 32000)

            # 2. 전처리 → 스펙트로그램 텐서
            tensor = preprocess(audio)       # (1, 1, 128, 128)

            # 3. 모델 추론
            probs = self.infer(tensor)       # [0.02, 0.91, 0.03, 0.04]

            # 4. 결과 해석
            idx   = int(np.argmax(probs))
            label = LABELS[idx]
            conf  = float(probs[idx])

            # 5. 경보 판단
            threshold = THRESHOLDS.get(label, 1.0)  # background는 임계값 없음
            now = time.time()

            if label != 'background' and conf >= threshold:
                if now - self._last_alert_time > self._alert_cooldown:
                    self._trigger_alert(label, conf)
                    self._last_alert_time = now
            # else: 아무 반응 없음

            # 6. 다음 추론까지 대기
            elapsed = time.time() - start
            sleep_time = INFER_EVERY - elapsed
            if sleep_time > 0:
                time.sleep(sleep_time)

    def _trigger_alert(self, label: str, confidence: float):
        msg = ALERT_MESSAGES.get(label, label)
        bar = "█" * int(confidence * 20)
        print(f"\r{msg}  신뢰도: [{bar:<20}] {confidence:.1%}")
        # 여기에 디스플레이 출력, LED, 알림음 등 연결 가능

    # ── 시작 / 종료 ────────────────────────────────────────────────
    def run(self):
        self._running = True
        infer_thread = threading.Thread(
            target=self._inference_loop, daemon=True
        )
        infer_thread.start()

        with sd.InputStream(
            device=self.device_id,
            samplerate=SR,
            channels=CHANNELS,
            blocksize=512,          # 32ms 블록
            dtype='int16',
            callback=self._audio_callback
        ):
            print("Ctrl+C 로 종료")
            try:
                while True:
                    time.sleep(1)
            except KeyboardInterrupt:
                print("\n종료")
        self._running = False


# ── 실행 진입점 ────────────────────────────────────────────────────
if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--model',  required=True,
                        help='.engine (TRT) 또는 .pt (PyTorch) 파일 경로')
    parser.add_argument('--device', type=int, default=None,
                        help='sounddevice 장치 번호 (기본값: 시스템 기본)')
    args = parser.parse_args()

    system = SoundAlertSystem(args.model, args.device)
    system.run()
```

## 경보가 울리는 조건 / 침묵하는 조건

| 상황 | 동작 |
|---|---|
| 경적 소리 + 신뢰도 ≥ 70% | "경적 감지!" 출력 |
| 사이렌 소리 + 신뢰도 ≥ 65% | "긴급차량 사이렌 감지!" 출력 |
| 배기음 + 신뢰도 ≥ 75% | "오토바이 접근!" 출력 |
| 조용한 상태 | 아무 반응 없음 |
| 도로 주행 소음, 바람 | 아무 반응 없음 |
| 경적이지만 신뢰도 < 70% | 아무 반응 없음 (임계값 미달) |
| 2초 이내 같은 경보 반복 | 한 번만 출력 (쿨다운) |

> 침묵의 핵심은 두 가지: **`label != 'background'`** 조건과 **신뢰도 임계값**. 이 둘을 모두 통과해야만 경보가 출력된다.
